cmake_minimum_required(VERSION 3.10)
project(cand3_exp CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenVINO REQUIRED COMPONENTS Runtime)
add_executable(cand3_exp main.cpp)
target_link_libraries(cand3_exp PRIVATE openvino::runtime)
add_executable(scale scale.cpp)
target_link_libraries(scale PRIVATE openvino::runtime)
add_executable(integ integ.cpp)
target_link_libraries(integ PRIVATE openvino::runtime)
add_executable(soak_probe soak_probe.cpp)
target_link_libraries(soak_probe PRIVATE openvino::runtime)
